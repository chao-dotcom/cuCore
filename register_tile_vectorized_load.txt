## 10b. Actually fixing the kernel: bigger register tile + vectorized loads + real pipelining

If Section 10 showed the double-buffered kernel landing close to (or worse than) the plain register-tiled one, that's a real signal, not noise -- and it's worth understanding *why* before assuming buffering helped nothing:

1. **The register tile was too small.** 4x4 outputs per thread means only 16 FMAs per k-step for 8 shared-memory reads (a compute:load ratio of 2). That's a low arithmetic intensity for a GPU this fast, so the kernel stays closer to memory-bound than compute-bound no matter how well you hide latency -- there's no long compute phase to hide anything *behind*.
2. **The "double buffering" in Section 10 wasn't real pipelining.** It prefetched next-tile data by reading *and immediately writing it to shared memory* at the top of the loop, before the compute phase. The `__syncthreads()` at the end of the iteration still forces every thread to wait for the slowest load before the next iteration's compute can begin -- there's no overlap window at all, just relabeled work with extra branch/lambda overhead layered on top (which is consistent with it measuring slightly *worse* at N=4096).

`matmul_v2` below fixes both:
- **8x8 register tile** (64 outputs/thread, 64 FMAs per k-step for 16 shared loads -> compute:load ratio of 4, double the old design's arithmetic intensity)
- **Vectorized `float4` loads** from global memory for both A and B tiles (fewer, wider memory transactions)
- **A tile stored transposed in shared memory** so the compute loop reads contiguous, conflict-free columns
- **True software-pipelined double buffering**: next-tile data is read into *registers* before the compute loop runs (so the load is in flight while the ALU-heavy 8x8 outer-product loop executes), and only written into the next shared-memory buffer *after* compute finishes -- one sync per iteration, with a real overlap window this time

Simplification for this benchmark: to keep the kernel readable, it assumes `N` is a multiple of 128 (true for all sizes in `matmul_sizes` above) and skips bounds-checking on the tile loads accordingly -- a production version would need edge-of-matrix masking for arbitrary `N`.

---

v2_source = r'''
// cuCore_v2_bench.cu
// A properly-pipelined, higher-arithmetic-intensity GEMM kernel, addressing the
// two real problems found in Section 10: an under-sized register tile (low
// compute:load ratio) and a "double-buffered" kernel that never actually
// overlapped memory latency with compute (it wrote to shared memory before
// syncing, so the sync still waited on the slow path every iteration).
//
// Fixes here:
//   - 8x8 per-thread register tile (128x128x8 block tile, 256 threads)
//   - float4 vectorized loads for both A and B tiles from global memory
//   - A tile stored transposed in shared memory for conflict-free column reads
//   - genuine software-pipelined double buffering: prefetch next tile into
//     registers *before* this tile's compute loop runs, defer the shared-memory
//     write until *after* compute, then sync once per iteration
//
// Simplification: assumes N is a multiple of 128 (true for every size this
// benchmark sweeps) so tile loads skip bounds checks.

#include "common/cuda_utils.cuh"
#include <cublas_v2.h>

void matmul_cpu(const float* A, const float* B, float* C, int N) {
    for (int i = 0; i < N; ++i)
        for (int j = 0; j < N; ++j) {
            double acc = 0.0;
            for (int k = 0; k < N; ++k) acc += (double)A[i*N+k] * B[k*N+j];
            C[i*N+j] = (float)acc;
        }
}

constexpr int BM = 128, BN = 128, BK = 8;
constexpr int TM = 8, TN = 8;
constexpr int THREADS = (BM / TM) * (BN / TN);  // 256

__global__ void matmul_v2_kernel(const float* __restrict__ A, const float* __restrict__ B,
                                  float* __restrict__ C, int N) {
    __shared__ float As[2][BK][BM];   // transposed: As[buf][k][m]
    __shared__ float Bs[2][BK][BN];   // natural:    Bs[buf][k][n]

    const int block_row = blockIdx.y * BM;
    const int block_col = blockIdx.x * BN;
    const int tid = threadIdx.x;
    const int tx = tid % (BN / TN);
    const int ty = tid / (BN / TN);

    // Loader index mappings (separate from the tx/ty compute mapping above).
    const int a_row = tid / 2, a_col_grp = tid % 2;        // A tile: 128 rows x 8 cols -> 256 float4s
    const int b_row = tid / 32, b_col_grp = tid % 32;      // B tile: 8 rows x 128 cols -> 256 float4s

    auto load_store_tile = [&](int k0, int buf) {
        float4 a4 = *reinterpret_cast<const float4*>(&A[(size_t)(block_row + a_row) * N + k0 + a_col_grp * 4]);
        As[buf][a_col_grp * 4 + 0][a_row] = a4.x;
        As[buf][a_col_grp * 4 + 1][a_row] = a4.y;
        As[buf][a_col_grp * 4 + 2][a_row] = a4.z;
        As[buf][a_col_grp * 4 + 3][a_row] = a4.w;

        float4 b4 = *reinterpret_cast<const float4*>(&B[(size_t)(k0 + b_row) * N + block_col + b_col_grp * 4]);
        *reinterpret_cast<float4*>(&Bs[buf][b_row][b_col_grp * 4]) = b4;
    };

    float acc[TM][TN] = {0.f};
    const int num_k_tiles = N / BK;

    int cur = 0;
    load_store_tile(0, cur);
    __syncthreads();

    for (int t = 0; t < num_k_tiles; ++t) {
        const int nxt = cur ^ 1;
        const bool has_next = (t + 1 < num_k_tiles);

        // Issue next tile's global reads into registers *now* -- these loads
        // are in flight while the compute loop below keeps the ALUs busy, so
        // their latency is hidden behind real work instead of behind a sync.
        float4 a4_next, b4_next;
        int k0_next = 0;
        if (has_next) {
            k0_next = (t + 1) * BK;
            a4_next = *reinterpret_cast<const float4*>(&A[(size_t)(block_row + a_row) * N + k0_next + a_col_grp * 4]);
            b4_next = *reinterpret_cast<const float4*>(&B[(size_t)(k0_next + b_row) * N + block_col + b_col_grp * 4]);
        }

#pragma unroll
        for (int k = 0; k < BK; ++k) {
            float a_reg[TM], b_reg[TN];
#pragma unroll
            for (int i = 0; i < TM; ++i) a_reg[i] = As[cur][k][ty * TM + i];
#pragma unroll
            for (int j = 0; j < TN; ++j) b_reg[j] = Bs[cur][k][tx * TN + j];
#pragma unroll
            for (int i = 0; i < TM; ++i)
#pragma unroll
                for (int j = 0; j < TN; ++j)
                    acc[i][j] += a_reg[i] * b_reg[j];
        }

        if (has_next) {
            // Only now (after compute) do we pay for the shared-memory write --
            // the global read latency above had this whole compute loop to hide in.
            As[nxt][a_col_grp * 4 + 0][a_row] = a4_next.x;
            As[nxt][a_col_grp * 4 + 1][a_row] = a4_next.y;
            As[nxt][a_col_grp * 4 + 2][a_row] = a4_next.z;
            As[nxt][a_col_grp * 4 + 3][a_row] = a4_next.w;
            *reinterpret_cast<float4*>(&Bs[nxt][b_row][b_col_grp * 4]) = b4_next;
        }
        __syncthreads();
        cur = nxt;
    }

#pragma unroll
    for (int i = 0; i < TM; ++i) {
        const int g_row = block_row + ty * TM + i;
#pragma unroll
        for (int j = 0; j < TN; ++j) {
            const int g_col = block_col + tx * TN + j;
            C[(size_t)g_row * N + g_col] = acc[i][j];
        }
    }
}

int main(int argc, char** argv) {
    const int N = (argc > 1) ? std::atoi(argv[1]) : 1024;
    if (N % BM != 0) {
        std::printf("This benchmark kernel assumes N is a multiple of %d; got N=%d. Skipping.\n", BM, N);
        return 1;
    }
    const bool check = (N <= 512);
    std::printf("matmul_v2 benchmark, N = %d\n", N);

    size_t bytes = (size_t)N * N * sizeof(float);
    std::vector<float> h_A(N*N), h_B(N*N), h_C_ref, h_C_gpu(N*N);
    fill_random(h_A, -1.f, 1.f, 31);
    fill_random(h_B, -1.f, 1.f, 32);

    if (check) {
        h_C_ref.resize(N*N);
        benchmark_cpu([&]() { matmul_cpu(h_A.data(), h_B.data(), h_C_ref.data(), N); }, 0, 1);
    }

    float *d_A, *d_B, *d_C;
    CUDA_CHECK(cudaMalloc(&d_A, bytes));
    CUDA_CHECK(cudaMalloc(&d_B, bytes));
    CUDA_CHECK(cudaMalloc(&d_C, bytes));
    CUDA_CHECK(cudaMemcpy(d_A, h_A.data(), bytes, cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemcpy(d_B, h_B.data(), bytes, cudaMemcpyHostToDevice));

    dim3 grid(N / BN, N / BM);
    dim3 block(THREADS);
    float v2_ms = benchmark_gpu([&]() {
        matmul_v2_kernel<<<grid, block>>>(d_A, d_B, d_C, N);
    }, 2, 10);
    CUDA_CHECK_LAST();

    bool ok = true;
    if (check) {
        CUDA_CHECK(cudaMemcpy(h_C_gpu.data(), d_C, bytes, cudaMemcpyDeviceToHost));
        ok = allclose(h_C_gpu.data(), h_C_ref.data(), (size_t)N*N, 1e-1, 1e-2);
    }

    cublasHandle_t handle; cublasCreate(&handle);
    const float alpha = 1.f, beta = 0.f;
    float cublas_ms = benchmark_gpu([&]() {
        cublasSgemm(handle, CUBLAS_OP_N, CUBLAS_OP_N, N, N, N, &alpha, d_B, N, d_A, N, &beta, d_C, N);
    }, 3, 20);
    cublasDestroy(handle);

    double flops = 2.0 * N * N * N;
    std::printf("\n================ matmul_v2 (GEMM), N=%d ================\n", N);
    std::printf("  V2 (8x8 reg, vectorized, pipelined): %10.4f ms  (%.2f GFLOP/s)\n", v2_ms, flops/1e9/(v2_ms/1e3));
    std::printf("  cuBLAS SGEMM                       : %10.4f ms  (%.2f GFLOP/s)\n", cublas_ms, flops/1e9/(cublas_ms/1e3));
    std::printf("  Fraction of cuBLAS achieved: %.1f%%\n", 100.0 * cublas_ms / v2_ms);
    std::printf("  Correctness: v2=%s\n", ok ? "PASS" : "FAIL");
    return ok ? 0 : 1;
}
'''

with open("kernel_v2_bench.cu", "w") as f:
    f.write(v2_source)

compile_kernel("matmul_v2", "kernel_v2_bench.cu", ["-lcublas"])
